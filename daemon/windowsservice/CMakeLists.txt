cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(kdeconnect-windows-service LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

if(NOT WIN32)
    message(FATAL_ERROR "The service host requires Windows")
endif()

add_executable(kdeconnect-service windowsservice.cpp desktopintegration.cpp)
target_compile_features(kdeconnect-service PRIVATE cxx_std_20)
set_target_properties(kdeconnect-service PROPERTIES WIN32_EXECUTABLE OFF AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(COMMAND kde_target_enable_exceptions)
    kde_target_enable_exceptions(kdeconnect-service PRIVATE)
endif()
target_compile_definitions(kdeconnect-service PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kdeconnect-service PRIVATE advapi32 userenv wtsapi32 shell32 ole32 propsys uuid)
# The privileged host must not load Qt or DLLs from a user-writable Scoop tree.
if(MSVC)
    set_property(TARGET kdeconnect-service PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_libraries(kdeconnect-service PRIVATE runtimeobject windowsapp)
elseif(MINGW)
    target_link_options(kdeconnect-service PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(service-test-worker testworker.cpp desktopintegration.cpp)
    target_compile_definitions(service-test-worker PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(service-test-worker PRIVATE advapi32 shell32 shlwapi ole32 propsys uuid)
    if(MSVC)
        set_property(TARGET service-test-worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
        target_link_options(service-test-worker PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
else()
    install(TARGETS kdeconnect-service ${KDECONNECT_INSTALL_TARGETS_DEFAULT_ARGS})
endif()
